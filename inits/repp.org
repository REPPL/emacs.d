  "Load a feature with fallback to source file if compiled version fails."
  (condition-case err
      (require feature)
    (error 
     (message "Compiled version failed for %s, trying source..." feature)
     (let ((source-file (concat (symbol-name feature) ".el")))
       (load source-file t)))))

  "Clean .elc files that commonly cause issues."
  (interactive)

     (format "find %s -name '*.elc' -size 0 -delete" 
             (shell-quote-argument elpa-dir)))

#+BEGIN_SRC emacs-lisp
(defun duplicate-current-line-or-region (arg)
  "Duplicates the current line or region ARG times.
If there's no region, the current line will be duplicated. However, if
there's a region, all lines that region covers will be duplicated."
  (interactive "p")
  (let (beg end (origin (point)))
    (if (and mark-active (> (point) (mark)))
        (exchange-point-and-mark))
    (setq beg (line-beginning-position))
    (if mark-active
        (exchange-point-and-mark))
    (setq end (line-end-position))
    (let ((region (buffer-substring-no-properties beg end)))
      (dotimes (i arg)
        (goto-char end)
        (newline)
        (insert region)
        (setq end (point)))
      (goto-char (+ origin (* (length region) arg) arg)))))
#+END_SRC

(defun academic-switch-template ()
  "Switch to a different template style.
Replaces preamble (everything before first heading) with template preamble,
preserving TITLE, SUBTITLE, AUTHOR, DATE, and all content after first heading.
Use this to change journal formats while keeping your manuscript content."
  (interactive)
  (let* ((template-name (completing-read "Switch to template: "
                                         (mapcar #'car academic-templates)
                                         nil t))
         (template-file (cdr (assoc template-name academic-templates)))
         (template-path (expand-file-name template-file
                                          (concat academic-production-dir "templates/org/")))
         template-preamble
         ;; Save current document metadata
         (saved-title nil)
         (saved-subtitle nil)
         (saved-author nil)
         (saved-date nil))
    (if (not (file-exists-p template-path))
        (message "Template not found: %s" template-path)
      ;; Extract metadata from current document
      (save-excursion
        (goto-char (point-min))
        (when (re-search-forward "^#\\+TITLE:[ \t]*\\(.+\\)$" nil t)
          (setq saved-title (match-string 1)))
        (goto-char (point-min))
        (when (re-search-forward "^#\\+SUBTITLE:[ \t]*\\(.+\\)$" nil t)
          (setq saved-subtitle (match-string 1)))
        (goto-char (point-min))
        (when (re-search-forward "^#\\+AUTHOR:[ \t]*\\(.+\\)$" nil t)
          (setq saved-author (match-string 1)))
        (goto-char (point-min))
        (when (re-search-forward "^#\\+DATE:[ \t]*\\(.+\\)$" nil t)
          (setq saved-date (match-string 1))))
      ;; Extract preamble from template (everything before first *)
      (with-temp-buffer
        (insert-file-contents template-path)
        (goto-char (point-min))
        (if (re-search-forward "^\\* " nil t)
            (setq template-preamble (buffer-substring (point-min) (match-beginning 0)))
          (setq template-preamble (buffer-string))))
      ;; Replace current preamble
      (save-excursion
        (goto-char (point-min))
        (let ((preamble-end (if (re-search-forward "^\\* " nil t)
                                (match-beginning 0)
                              (point-max))))
          (delete-region (point-min) preamble-end)
          (goto-char (point-min))
          (insert template-preamble)))
      ;; Restore saved metadata
      (save-excursion
        (when saved-title
          (goto-char (point-min))
          (when (re-search-forward "^#\\+TITLE:.*$" nil t)
            (replace-match (concat "#+TITLE: " saved-title))))
        (when saved-subtitle
          (goto-char (point-min))
          (if (re-search-forward "^#\\+SUBTITLE:.*$" nil t)
              (replace-match (concat "#+SUBTITLE: " saved-subtitle))
            ;; Insert subtitle after title if template doesn't have one
            (goto-char (point-min))
            (when (re-search-forward "^#\\+TITLE:.*$" nil t)
              (end-of-line)
              (insert "\n#+SUBTITLE: " saved-subtitle))))
        (when saved-author
          (goto-char (point-min))
          (when (re-search-forward "^#\\+AUTHOR:.*$" nil t)
            (replace-match (concat "#+AUTHOR: " saved-author))))
        (when saved-date
          (goto-char (point-min))
          (when (re-search-forward "^#\\+DATE:.*$" nil t)
            (replace-match (concat "#+DATE: " saved-date)))))
      ;; Fix CSL path to be relative to manuscript location
      (when buffer-file-name
        (save-excursion
          (goto-char (point-min))
          (when (re-search-forward "^#\\+CITE_EXPORT:[ \t]*csl[ \t]+\\(\\.\\./csl/[^ \t\n]+\\)" nil t)
            (let* ((csl-file (file-name-nondirectory (match-string 1)))
                   (manuscript-dir (file-name-directory buffer-file-name))
                   (csl-dir (concat academic-production-dir "csl/"))
                   (rel-path (file-relative-name csl-dir manuscript-dir)))
              (replace-match (concat "#+CITE_EXPORT: csl " rel-path csl-file))))))
      (message "Switched to template: %s (preserved title/author/date/content)" template-name))))

(defun academic-inject-base-preamble ()
  "Inject base preamble into the current buffer after LATEX_COMPILER line.
The base preamble contains common packages (booktabs, caption, hyperref, etc.)
that are shared across all templates."
  (when (file-exists-p academic-base-preamble-file)
    (save-excursion
      (goto-char (point-min))
      ;; Check if base preamble already injected
      (unless (re-search-forward "# Common packages" nil t)
        ;; Find insertion point: after LATEX_COMPILER or before first LATEX_HEADER
        (goto-char (point-min))
        (let ((insert-point nil)
              (base-content nil))
          ;; Read base preamble content (skip comment lines at top)
          (with-temp-buffer
            (insert-file-contents academic-base-preamble-file)

                        (looking-at "^#[^+]"))

          ;; Find insertion point
          (goto-char (point-min))
          (if (re-search-forward "^#\\+LATEX_COMPILER:.*$" nil t)

            (goto-char (point-min))
            (when (re-search-forward "^#\\+LATEX_HEADER:" nil t)
              (beginning-of-line)
              (setq insert-point (point))))
          ;; Insert base preamble
          (when (and insert-point base-content)
            (goto-char insert-point)
            (insert "\n" base-content)))))))

;; Convert tables with :span t to table* for two-column layouts
;; Strategy: Track table positions with :span t, then convert those specific tables

(defvar academic--span-table-positions nil
  "List of table positions (1-indexed) that should be converted to table*.")

(defun academic-track-span-tables (backend)
  "Track which tables have :span t attribute before export."
  (when (org-export-derived-backend-p backend 'latex)
    (setq academic--span-table-positions nil)
    (let ((table-num 0))
      (goto-char (point-min))
      ;; Find all tables and note which have :span t
      (while (re-search-forward "^#\\+\\(CAPTION\\|NAME\\|ATTR_LATEX\\):" nil t)
        (let ((block-start (line-beginning-position))
              (has-span nil)
              (is-table nil))
          ;; Look at this block of #+LINES
          (save-excursion
            (goto-char block-start)
            (while (looking-at "^#\\+")
              (when (looking-at "^#\\+ATTR_LATEX:.*:span t")
                (setq has-span t))
              (forward-line 1))
            ;; Check if followed by a table
            (when (looking-at "^|")
              (setq is-table t)))
          ;; If this is a table, count it
          (when is-table
            (setq table-num (1+ table-num))
            (when has-span
              (push table-num academic--span-table-positions)))
          ;; Move past this block
          (while (and (not (eobp)) (looking-at "^#\\+\\|^|"))
            (forward-line 1)))))
    (setq academic--span-table-positions (nreverse academic--span-table-positions))))

(add-hook 'org-export-before-processing-hook #'academic-track-span-tables)

(defun academic-convert-tables-to-starred (text backend info)
  "Convert specific table environments to table* based on tracked positions."
  (if (or (not (org-export-derived-backend-p backend 'latex))
          (null academic--span-table-positions))
      text
    (with-temp-buffer
      (insert text)
      (goto-char (point-min))
      (let ((table-num 0))
        (while (re-search-forward "\\\\begin{table}" nil t)
          (setq table-num (1+ table-num))
          (when (member table-num academic--span-table-positions)
            (replace-match "\\\\begin{table*}")
            (when (re-search-forward "\\\\end{table}" nil t)
              (replace-match "\\\\end{table*}")))))
      (buffer-string))))

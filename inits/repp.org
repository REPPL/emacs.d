* Setup wizard

First-run interactive wizard that detects missing system tools and offers to
install them. Can be re-invoked any time with =M-x ar-setup-wizard=.
Sentinel file ~~/.emacs.d/.setup-done~ prevents re-prompting on subsequent
launches.

#+BEGIN_SRC emacs-lisp
(defgroup ar-setup nil
  "First-run setup wizard for this Emacs configuration."
  :group 'convenience)

(defcustom ar-setup-venv-root "~/.venvs/"
  "Root directory for Python virtual environments managed by the wizard."
  :type 'directory
  :group 'ar-setup)

(defcustom ar-setup-sentinel
  (expand-file-name ".setup-done" user-emacs-directory)
  "File whose existence indicates every wizard category completed successfully.
While missing, the wizard auto-prompts on Emacs startup."
  :type 'file
  :group 'ar-setup)

(defconst ar-setup--log-buffer "*ar-setup-log*")

(defun ar-setup--have-p (binary)
  "Return non-nil if BINARY is on PATH."
  (executable-find binary))

(defun ar-setup--log (fmt &rest args)
  "Append a formatted line to the setup log buffer."
  (with-current-buffer (get-buffer-create ar-setup--log-buffer)
    (goto-char (point-max))
    (insert (apply #'format fmt args) "\n")))

(defun ar-setup--run (command)
  "Run COMMAND in a shell, streaming output into the log buffer.
Returns the exit code."
  (ar-setup--log "$ %s" command)
  (let ((exit (call-process-shell-command
               command nil ar-setup--log-buffer t)))
    (ar-setup--log "[exit %d]" exit)
    exit))

(defun ar-setup--confirm (prompt)
  "Yes/no/skip-all prompt in the minibuffer (never a GUI dialog).
Returns `yes', `no', or `quit'. Logs the prompt and the user's answer."
  (ar-setup--log "? %s" prompt)
  (let* ((use-dialog-box nil)
         (last-nonmenu-event t)
         (c (read-char-choice (concat prompt " [y/n/q] ") '(?y ?n ?q)))
         (answer (pcase c (?y 'yes) (?n 'no) (?q 'quit))))
    (ar-setup--log "  -> %s" answer)
    answer))

(defun ar-setup--step (name check-fn install-cmd)
  "If CHECK-FN returns non-nil, skip NAME. Otherwise prompt and on yes
run INSTALL-CMD (a shell string).
Returns `ok', `declined', or `quit'."
  (cond
   ((funcall check-fn)
    (ar-setup--log "[ok ] %s already installed" name)
    'ok)
   (t
    (ar-setup--log "[miss] %s not found" name)
    (pcase (ar-setup--confirm
            (format "Install %s? Will run:\n    %s\n" name install-cmd))
      ('yes  (if (zerop (ar-setup--run install-cmd)) 'ok 'declined))
      ('no   (ar-setup--log "[skip] %s declined by user" name) 'declined)
      ('quit 'quit)))))

(defun ar-setup--brew-available-p ()
  (ar-setup--have-p "brew"))

;; --- Per-category "is anything missing?" predicates ---
;; These let the wizard auto-skip the outer prompt entirely when a category
;; has nothing to install. Each predicate returns non-nil if the category
;; needs user attention (i.e. at least one tool is missing).

(defun ar-setup--need-core-p ()
  (not (and (ar-setup--have-p "aspell")
            (ar-setup--have-p "pandoc")
            (ar-setup--have-p "pipx"))))

(defun ar-setup--need-python-lsp-p ()
  (not (ar-setup--have-p "pylsp")))

(defun ar-setup--need-python-tooling-p ()
  (not (and (ar-setup--have-p "black")
            (ar-setup--have-p "isort")
            (ar-setup--have-p "flake8"))))

(defun ar-setup--need-treesit-p ()
  (and (fboundp 'treesit-language-available-p)
       (not (treesit-language-available-p 'python))))

(defun ar-setup--need-default-venv-p ()
  (let* ((venv (expand-file-name "default"
                                 (expand-file-name ar-setup-venv-root)))
         (pip  (expand-file-name "bin/pip" venv)))
    (not (file-exists-p pip))))

(defun ar-setup--check-core ()
  "Check core tools: aspell, pandoc, pipx."
  (catch 'quit
    (unless (ar-setup--brew-available-p)
      (ar-setup--log "[warn] Homebrew not found; skipping brew-managed tools")
      (throw 'quit 'declined))
    (dolist (pkg '(("aspell" . "brew install aspell")
                   ("pandoc" . "brew install pandoc")
                   ("pipx"   . "brew install pipx")))
      (let ((bin (car pkg)) (cmd (cdr pkg)))
        (when (eq 'quit (ar-setup--step bin
                                        (lambda () (ar-setup--have-p bin))
                                        cmd))
          (throw 'quit 'quit))))
    'ok))

(defun ar-setup--check-python-lsp ()
  "Check Python LSP. Offers pylsp via pipx."
  (catch 'quit
    (unless (ar-setup--have-p "pipx")
      (ar-setup--log "[warn] pipx not installed; cannot install pylsp")
      (throw 'quit 'declined))
    (ar-setup--step "python-lsp-server (pylsp)"
                    (lambda () (ar-setup--have-p "pylsp"))
                    "pipx install 'python-lsp-server[all]'")))

(defun ar-setup--check-python-tooling ()
  "Check global Python dev tools via pipx."
  (catch 'quit
    (unless (ar-setup--have-p "pipx")
      (ar-setup--log "[warn] pipx not installed; cannot install Python tooling")
      (throw 'quit 'declined))
    (dolist (tool '("black" "isort" "flake8"))
      (when (eq 'quit (ar-setup--step
                       tool
                       (lambda () (ar-setup--have-p tool))
                       (format "pipx install %s" tool)))
        (throw 'quit 'quit)))
    'ok))

(defun ar-setup--check-treesit ()
  "Offer to install tree-sitter grammars used by the config.
Returns `ok' only if all required grammars are present (or were
successfully installed). Provides a default `treesit-language-source-alist'
recipe for languages the wizard manages if the user has not configured
one."
  (catch 'quit
    (unless (fboundp 'treesit-language-available-p)
      (ar-setup--log "[warn] Emacs lacks tree-sitter support; skipping")
      (throw 'quit 'declined))
    ;; Provide a recipe for python if none is already registered.
    ;; Pinned to v0.21.0 (tree-sitter ABI 14) for compatibility with the
    ;; libtree-sitter shipped in Emacs 30.x. master uses ABI 15 which
    ;; triggers a version-mismatch warning and disables highlighting.
    (unless (assq 'python (and (boundp 'treesit-language-source-alist)
                               treesit-language-source-alist))
      (add-to-list 'treesit-language-source-alist
                   '(python "https://github.com/tree-sitter/tree-sitter-python"
                            "v0.21.0")))
    (let ((all-present t))
      (dolist (lang '(python))
        (cond
         ((treesit-language-available-p lang)
          (ar-setup--log "[ok ] tree-sitter %s already available" lang))
         (t
          (pcase (ar-setup--confirm
                  (format "Install tree-sitter grammar for %s? (clones from GitHub, compiles)" lang))
            ('yes (condition-case err
                      (progn (treesit-install-language-grammar lang)
                             (ar-setup--log "[ok ] tree-sitter %s installed" lang))
                    (error (ar-setup--log "[err] %s grammar install failed: %S" lang err)
                           (setq all-present nil))))
            ('no  (ar-setup--log "[skip] tree-sitter %s declined" lang)
                  (setq all-present nil))
            ('quit (throw 'quit 'quit))))))
      (if all-present 'ok 'declined))))

(defun ar-setup--check-default-venv ()
  "Offer to create a default Python venv under ~ar-setup-venv-root~.
Returns `ok' if the venv already exists or was successfully created.
Returns `declined' if user said no or venv creation failed."
  (catch 'quit
    (let* ((root (expand-file-name ar-setup-venv-root))
           (venv (expand-file-name "default" root))
           (pip  (expand-file-name "bin/pip" venv))
           (python (ar-setup--have-p "python3")))
      (cond
       ((not python)
        (ar-setup--log "[warn] python3 not on PATH; cannot create venv")
        'declined)
       ((file-exists-p pip)
        (ar-setup--log "[ok ] default venv exists at %s" venv)
        'ok)
       (t
        (pcase (ar-setup--confirm
                (format "Create default Python venv at %s and install dev tools?" venv))
          ('yes
           (unless (file-directory-p root) (make-directory root t))
           (let ((rc1 (ar-setup--run (format "%s -m venv %s"
                                             (shell-quote-argument python)
                                             (shell-quote-argument venv))))
                 (rc2 (ar-setup--run
                       (format "%s install --upgrade pip black isort flake8 'python-lsp-server[all]' ipython"
                               (shell-quote-argument pip)))))
             (if (and (zerop rc1) (zerop rc2)) 'ok 'declined)))
          ('no   (ar-setup--log "[skip] default venv declined") 'declined)
          ('quit (throw 'quit 'quit))))))))

(defun ar-setup-wizard (&optional force)
  "Run the interactive first-run setup wizard.
With prefix arg FORCE, run even if the sentinel file exists."
  (interactive "P")
  (if (and (not force) (file-exists-p ar-setup-sentinel))
      (message "Setup already complete (%s). Use C-u M-x ar-setup-wizard to re-run."
               ar-setup-sentinel)
    (ar-setup--wizard-body)))

(defconst ar-setup--steps
  '(("Core tools (aspell, pandoc, pipx)"
     ar-setup--need-core-p           ar-setup--check-core)
    ("Python LSP (pylsp)"
     ar-setup--need-python-lsp-p     ar-setup--check-python-lsp)
    ("Python tooling (black, isort, flake8)"
     ar-setup--need-python-tooling-p ar-setup--check-python-tooling)
    ("Default Python venv (~/.venvs/default)"
     ar-setup--need-default-venv-p   ar-setup--check-default-venv))
  "Ordered list of (LABEL PREDICATE FUNCTION) wizard categories.
PREDICATE returns non-nil when the category needs user attention.
If PREDICATE returns nil, the category is silently marked `ok' and the
outer prompt is skipped entirely.

Tree-sitter grammar installation was removed after an ABI mismatch
crashed Emacs on opening .py files. Install grammars manually if
needed.")

(defun ar-setup--wizard-body ()
  "Internal body of the wizard. Fully inline: prompts via the minibuffer,
status messages via `message' (visible in *Messages*). No buffer popups,
no window splits.

Tracks per-category status; writes the sentinel only if every category
returned `ok' (i.e. nothing was declined or aborted). On next launch,
any category not marked `ok' is offered again. Full log preserved
silently in *ar-setup-log* for inspection via `M-x switch-to-buffer'."
  (let ((use-dialog-box nil)
        (last-nonmenu-event t))
    (with-current-buffer (get-buffer-create ar-setup--log-buffer)
      (erase-buffer)
      (insert (format-time-string "[ar-setup %Y-%m-%d %H:%M:%S]\n")))
    (ar-setup--log "PATH: %s" (getenv "PATH"))
    (ar-setup--log "exec-path-from-shell-arguments: %S"
                   (and (boundp 'exec-path-from-shell-arguments)
                        exec-path-from-shell-arguments))
    (dolist (bin '("aspell" "pandoc" "pipx" "pylsp"
                   "black" "isort" "flake8" "python3"))
      (ar-setup--log "  %s: %s" bin (or (executable-find bin) "(not found)")))
    ;; Decide upfront which categories need attention; auto-skip the rest.
    (let* ((all-steps ar-setup--steps)
           (needed (cl-remove-if-not (lambda (s) (funcall (nth 1 s))) all-steps))
           (auto-ok (cl-remove-if    (lambda (s) (funcall (nth 1 s))) all-steps)))
      (dolist (s auto-ok)
        (ar-setup--log "[ok ] %s — nothing to install" (car s))
        (message "  [ok] %s" (car s)))
      (cond
       ((null needed)
        (with-temp-buffer (write-file ar-setup-sentinel))
        (message "Setup: nothing to do — all %d categories already complete."
                 (length all-steps)))
       (t
        (message "Setup wizard: %d categor%s need attention. y=install, n=skip, q=quit."
                 (length needed) (if (= 1 (length needed)) "y" "ies"))))
      (let ((results (mapcar (lambda (s) (cons (car s) 'ok)) auto-ok))
            (aborted nil))
        (catch 'abort
          (dolist (step needed)
            (let* ((label (nth 0 step))
                   (fn    (nth 2 step))
                   (status
                    (pcase (ar-setup--confirm (format "Setup: %s?" label))
                      ('yes  (let ((r (funcall fn)))
                               (when (eq r 'quit) (setq aborted t) (throw 'abort nil))
                               r))
                      ('no   (ar-setup--log "[skip] section: %s" label) 'declined)
                      ('quit (setq aborted t) (throw 'abort nil)))))
              (push (cons label status) results)
              (message "  [%s] %s" status label))))
      (setq results (nreverse results))
      (let* ((all-ok (and (not aborted)
                          (cl-every (lambda (r) (eq (cdr r) 'ok)) results)))
             (pending (cl-remove-if (lambda (r) (eq (cdr r) 'ok)) results)))
        (cond
         (aborted
          (message "Setup wizard aborted. Re-run with M-x ar-setup-wizard."))
         (all-ok
          (with-temp-buffer (write-file ar-setup-sentinel))
          (message "Setup complete. All %d categories ok. Log: M-x switch-to-buffer %s"
                   (length results) ar-setup--log-buffer))
         (t
          (message "Setup partial: %d pending (%s). Will re-prompt next launch."
                   (length pending)
                   (mapconcat #'car pending ", ")))))))))

;; The wizard is opt-in: never auto-prompt. If setup has not been
;; completed (sentinel missing), just print a passive one-line note in
;; the echo area pointing at `M-x ar-setup-wizard'. Runs after startup
;; settles so the note is not overwritten by other startup messages.
(add-hook 'emacs-startup-hook
          (lambda ()
            (unless (file-exists-p ar-setup-sentinel)
              (run-with-idle-timer
               1 nil
               (lambda ()
                 (message "Tip: optional tooling setup available via M-x ar-setup-wizard"))))))

*** DONE Duplicate line

#+BEGIN_SRC emacs-lisp
(defun duplicate-current-line-or-region (arg)
  "Duplicates the current line or region ARG times.
If there's no region, the current line will be duplicated. However, if
there's a region, all lines that region covers will be duplicated."
  (interactive "p")
  (let (beg end (origin (point)))
    (if (and mark-active (> (point) (mark)))
        (exchange-point-and-mark))
    (setq beg (line-beginning-position))
    (if mark-active
        (exchange-point-and-mark))
    (setq end (line-end-position))
    (let ((region (buffer-substring-no-properties beg end)))
      (dotimes (i arg)
        (goto-char end)
        (newline)
        (insert region)
        (setq end (point)))
      (goto-char (+ origin (* (length region) arg) arg)))))
#+END_SRC

*** DONE Org-style export menu (via pandoc)

A transient dispatcher on ~C-c C-e~ — mirroring ~org-export-dispatch~ —
that exports the current Markdown file to HTML, PDF, Org, ASCII or Word
through pandoc. Everything is deferred until markdown-mode loads, so it
adds nothing to startup. PDF additionally needs a LaTeX engine (for
example BasicTeX or MacTeX).

#+BEGIN_SRC emacs-lisp
;; Make the transient-define-prefix macro available when this file is
;; byte-compiled; transient itself is required lazily at runtime below.
(eval-when-compile (require 'transient))

(with-eval-after-load 'markdown-mode
  (require 'transient)

  (defun ar-markdown--export (label extension &rest pandoc-args)
    "Export the current Markdown buffer to EXTENSION via pandoc.
LABEL names the format in messages; PANDOC-ARGS are extra pandoc flags.
The output file is written next to the source."
    (unless (buffer-file-name)
      (user-error "Save this buffer to a file before exporting"))
    (unless (executable-find "pandoc")
      (user-error "pandoc is not installed (the installer offers to install it)"))
    (save-buffer)
    (let* ((src (buffer-file-name))
           (out (concat (file-name-sans-extension src) "." extension))
           (log (get-buffer-create "*markdown-export*"))
           (args (append (list src "-o" out) pandoc-args)))
      (with-current-buffer log (erase-buffer))
      (if (eq 0 (apply #'call-process "pandoc" nil log nil args))
          (message "%s exported to %s" label (file-name-nondirectory out))
        (display-buffer log)
        (user-error "pandoc failed exporting %s (see *markdown-export*)" label))))

  (defun ar-markdown-export-html ()
    "Export the current Markdown buffer to standalone HTML."

    "Export the current Markdown buffer to PDF (needs a LaTeX engine)."
    (interactive) (ar-markdown--export "PDF" "pdf"))
  (defun ar-markdown-export-org ()
    "Export the current Markdown buffer to Org."

  (defun ar-markdown-export-ascii ()
    "Export the current Markdown buffer to plain-text ASCII."
    (interactive) (ar-markdown--export "ASCII" "txt" "-t" "plain"))

;; Python workflow helper functions
(defun python-setup-windows ()
  "Set up ideal Python development layout: code above, shell below."
  (interactive)
  (delete-other-windows)
  (split-window-below)
  (other-window 1)
  (if (get-buffer "*Python*")
      (switch-to-buffer "*Python*")
    (python-shell-switch-to-shell))
  (other-window 1))

(defun python-run-buffer-and-show ()
  "Send buffer to Python shell and switch to shell to see output."
  (interactive)
  (python-shell-send-buffer)
  (python-shell-switch-to-shell))

(defun python-run-region-and-show ()
  "Send selected region to Python shell and switch to shell to see output."
  (interactive)
  (if (use-region-p)
      (progn
        (python-shell-send-region (region-beginning) (region-end))
        (python-shell-switch-to-shell))
    (message "No region selected")))

;; Auto-start a Python subprocess if none exists, then send the buffer.
;; Replaces python-shell-send-buffer on C-c C-c so the "Start a Python
;; process first with M-x run-python" nag goes away.
(defun ar-python-send-buffer-or-start ()
  "Send current buffer to *Python*, starting it first if needed."
  (interactive)
  (unless (python-shell-get-process)
    (save-window-excursion (run-python nil nil t))
    (sit-for 0.2))
  (python-shell-send-buffer))

(defun dired-sidebar ()
  "Open dired in a side window for project navigation."
  (interactive)
  (let ((project-root (or (and (fboundp 'projectile-project-root)
